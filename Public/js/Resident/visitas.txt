document.addEventListener("DOMContentLoaded", () => {
  const forms = document.querySelectorAll(".form-estatus");

  forms.forEach(form => {
    form.addEventListener("submit", (e) => {
      const select = form.querySelector("select");
      if (!select.value) {
        e.preventDefault();
        alert("Por favor selecciona un nuevo estatus antes de actualizar.");
      }
    });
  });
});
